# XML stores the authoritative SID. The COM property may resolve that SID to
# a local account alias; compare its translated SID, not the display spelling.
function Assert-CompanionTaskOwner([object]$Task) {
  $xml = [xml][string]$Task.Xml
  $principals = @($xml.SelectNodes('//*[local-name()="Principals"]/*[local-name()="Principal"]'))
  if ($principals.Count -ne 1) { throw 'TASK_XML_PRINCIPAL_COUNT' }
  $userIds = @($principals[0].SelectNodes('*[local-name()="UserId"]'))
  if ($userIds.Count -ne 1 -or $userIds[0].InnerText -cne $script:CompanionOwnerSid) { throw 'TASK_XML_OWNER_SID' }
  $reported = [string]$Task.Definition.Principal.UserId
  if ($reported -match '^S-1-') { $resolved = [Security.Principal.SecurityIdentifier]::new($reported).Value }
  else { $resolved = [Security.Principal.NTAccount]::new($reported).Translate([Security.Principal.SecurityIdentifier]).Value }
  if ($resolved -cne $script:CompanionOwnerSid) { throw 'TASK_RESOLVED_OWNER_SID' }
  $security = [Security.AccessControl.RawSecurityDescriptor]::new([string]$Task.GetSecurityDescriptor(5))
  if ($security.Owner.Value -cne $script:CompanionOwnerSid -or $null -eq $security.DiscretionaryAcl -or $security.DiscretionaryAcl.Count -lt 1) { throw 'TASK_SECURITY_OWNER_OR_EMPTY_ACL' }
  foreach ($ace in $security.DiscretionaryAcl) {
    if ($ace -isnot [Security.AccessControl.CommonAce] -or $ace.AceQualifier -ne [Security.AccessControl.AceQualifier]::AccessAllowed -or $ace.SecurityIdentifier.Value -cne $script:CompanionOwnerSid) { throw 'TASK_SECURITY_OTHER_OWNER_OR_DENY' }
  }
  return @{xmlOwnerSid=$script:CompanionOwnerSid;resolvedOwnerSid=$resolved;reportedPrincipal=$reported;taskAclOwnerOnly=$true}
}
