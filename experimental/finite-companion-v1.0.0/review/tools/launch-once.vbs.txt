Function Quote(value)
  If InStr(value, Chr(34)) <> 0 Or InStr(value, vbCr) <> 0 Or InStr(value, vbLf) <> 0 Then WScript.Quit 105
  Quote = Chr(34) & value & Chr(34)
End Function
Function Matches(pattern, value)
  Dim re
  Set re = New RegExp
  re.Pattern = pattern
  re.IgnoreCase = False
  re.Global = False
  Matches = re.Test(value)
End Function
Function PrivateFolder(value)
  Dim item
  PrivateFolder = False
  On Error Resume Next
  Set item = fs.GetFolder(value)
  If Err.Number <> 0 Then Err.Clear: Exit Function
  If (item.Attributes And 1024) <> 0 Then Exit Function
  PrivateFolder = PrivateAcl(value, False)
  On Error GoTo 0
End Function
Function PrivateSealedFile(value)
  Dim item
  PrivateSealedFile = False
  On Error Resume Next
  Set item = fs.GetFile(value)
  If Err.Number <> 0 Then Err.Clear: Exit Function
  If (item.Attributes And 1024) <> 0 Or (item.Attributes And 1) = 0 Then Exit Function
  If value = grantFile And (item.Size < 1 Or item.Size > 512) Then Exit Function
  PrivateSealedFile = PrivateAcl(value, True)
  On Error GoTo 0
End Function
Function PrivateAcl(value, readOnly)
  Dim setting, descriptor, result, ace, ownerCount, pathValue
  PrivateAcl = False
  pathValue = Replace(Replace(value, "\", "\\"), "'", "\'")
  On Error Resume Next
  Set setting = wmi.Get("Win32_LogicalFileSecuritySetting.Path='" & pathValue & "'")
  result = setting.GetSecurityDescriptor(descriptor)
  If Err.Number <> 0 Or result <> 0 Then Err.Clear: Exit Function
  If descriptor.Owner.SIDString <> ExpectedSid Then Exit Function
  If (descriptor.ControlFlags And 4096) = 0 Then Exit Function
  ownerCount = 0
  For Each ace In descriptor.DACL
    If ace.AceType <> 0 Or ace.Trustee.SIDString <> ExpectedSid Then Exit Function
    If readOnly And (ace.AccessMask And &H500D0156) <> 0 Then Exit Function
    ownerCount = ownerCount + 1
  Next
  If Err.Number <> 0 Then Err.Clear: Exit Function
  PrivateAcl = (ownerCount > 0)
  On Error GoTo 0
End Function
