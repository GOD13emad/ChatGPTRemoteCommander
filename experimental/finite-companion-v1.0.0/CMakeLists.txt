cmake_minimum_required(VERSION 3.21)
project(commander_job_guard LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
add_executable(commander-job-guard native/job-guard.cc)
target_compile_features(commander-job-guard PRIVATE cxx_std_17)
target_compile_options(commander-job-guard PRIVATE /utf-8 /W4)
target_link_libraries(commander-job-guard PRIVATE advapi32)
